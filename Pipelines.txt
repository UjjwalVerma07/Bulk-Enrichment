#!/bin/bash

# Kafka bootstrap server
BOOTSTRAP_SERVER="localhost:9092"

# List of topics to create
TOPICS=(
  "pipeline-config"
  "pipeline-progress"
  "enrich-input-topic"
  "email-validation-output"
  "reverse-geocode-output"
  "enrich-output-topic"
  "enrich-input-topic1"
  "email-validation-output1"
  "reverse-geocode-output1"
  "enrich-output-topic1"
)

# Loop through each topic and create it
for TOPIC in "${TOPICS[@]}"; do
  kafka-topics.sh --create \
    --topic "$TOPIC" \
    --bootstrap-server $BOOTSTRAP_SERVER \
    --partitions 1 \
    --replication-factor 1 \
    2>/dev/null

  if [ $? -eq 0 ]; then
    echo "Created topic: $TOPIC"
  else
    echo "Topic $TOPIC already exists or failed to create"
  fi
done
